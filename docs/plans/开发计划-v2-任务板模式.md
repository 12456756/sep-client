## 认证架构（不变）

SEP 使用**三层 Token 架构**：

| Token 类型 | 用途 | 有效期 | 获取方式 | 存储位置 |
|-----------|------|--------|---------|---------|
| 通用 accessToken | 获取实例列表 | 1小时 | 邮箱密码登录 | 内存 |
| refreshToken | 换取实例令牌 | 30天 | 邮箱密码登录 | safeStorage 加密 |
| 实例级 instanceToken | 调用 AI 网关 | **15分钟** | refreshToken + instanceId | 内存 |

interface Task {
  id: string;                    // UUID
  title: string;                 // 任务标题
  prompt: string;                // 完整的任务描述（给 AI 的）
  status: TaskStatus;            // 任务状态

  createdAt: number;             // 创建时间戳
  startedAt: number | null;      // 开始时间
  completedAt: number | null;    // 完成时间
  error: string | null;          // 错误信息
  files: string[];               // 生成的文件列表
  logs: TaskLog[];               // 执行日志
}

enum TaskStatus {
  PENDING = 'pending',           // 待执行
  RUNNING = 'running',           // 执行中
  WAITING_APPROVAL = 'waiting_approval',  // 等待审批
  PAUSED = 'paused',            // 已暂停
  COMPLETED = 'completed',       // 已完成
  FAILED = 'failed',            // 失败

class TaskManager {
  private tasks: Map<string, Task> = new Map();
  private currentTaskId: string | null = null;
  
  // 创建任务
  createTask(title: string, prompt: string, workDir?: string): Task;
  
  // 执行任务
  async executeTask(taskId: string): Promise<void>;
  
  // 暂停任务
  pauseTask(taskId: string): void;
  
  // 取消任务
  cancelTask(taskId: string): void;
  
  // 更新任务状态
  updateTaskStatus(taskId: string, status: TaskStatus): void;
  
  // 添加任务日志
  addTaskLog(taskId: string, log: TaskLog): void;
  
  // 获取所有任务

  // 获取任务详情

}

#### 3.2 Pi Session 集成

**文件**: `electron/pi-host.ts` (修改)

**关键变化**:
```typescript
class PiHost {
  private taskManager: TaskManager;
  
  // 执行任务（而不是发送单个 prompt）
  async executeTask(task: Task): Promise<void> {
    if (!this.session) throw new Error('Session not started');
    
    // 设置当前任务上下文
    this.currentTask = task;
    
    // 发送 prompt
    this.session.prompt(task.prompt);
  }
  
  // 监听事件，映射到任务状态
  private setupEventHandlers() {
    this.session.subscribe((event) => {
      switch (event.type) {
        case 'agent_start':
          this.taskManager.updateTaskStatus(this.currentTask.id, 'running');
          this.taskManager.addTaskLog(this.currentTask.id, {
            timestamp: Date.now(),
            message: 'AI 开始处理任务',
          });
          break;
          
        case 'message_update':
          // 可选：记录 AI 的思考过程
          break;
          
        case 'tool_execution_start':
          this.taskManager.addTaskLog(this.currentTask.id, {
            timestamp: Date.now(),
            message: `执行工具: ${event.toolName}`,
          });
          break;
          
        case 'tool_execution_end':
          this.taskManager.addTaskLog(this.currentTask.id, {
            timestamp: Date.now(),
            message: `工具执行完成: ${event.toolName}`,
          });
          break;
          
        case 'agent_settled':
          this.taskManager.updateTaskStatus(this.currentTask.id, 'completed');
          this.taskManager.addTaskLog(this.currentTask.id, {
            timestamp: Date.now(),
            message: '任务完成',
          });
          // 发送通知
          this.sendNotification('任务完成', this.currentTask.title);
          break;
          
        case 'agent_end':
          if (event.error) {
            this.taskManager.updateTaskStatus(this.currentTask.id, 'failed');
            this.taskManager.addTaskLog(this.currentTask.id, {
              timestamp: Date.now(),
              message: `任务失败: ${event.error}`,
              level: 'error',
            });
          }
          break;
      }
      
      // 转发事件到 renderer（实时更新 UI）
      this.config.onEvent({
        type: 'task:updated',
        taskId: this.currentTask.id,
      });
    });
  }
}
```

#### 3.3 工作目录权限检查

**文件**: `pi-extension/guard.ts` (修改)

```typescript
export function buildPermissionGuard(config: GuardConfig): ExtensionFactory {
  return (pi) => {
    pi.on('tool_call', async (event: ToolCallEvent): Promise<ToolCallEventResult> => {
      const toolName = event.toolName;
      
      // 获取当前任务的工作目录
      const workDir = config.getCurrentWorkDir();
      
      // 1. 只读工具 → 自动允许
      if (READ_ONLY_TOOLS.has(toolName)) {
        return { block: false };
      }
      
      // 2. 文件操作工具 → 检查是否在工作目录内
      if (toolName === 'write' || toolName === 'edit') {
        const targetPath = event.args.file_path;
        
        if (workDir && isPathInside(targetPath, workDir)) {
          // 在工作目录内 → 自动允许
          return { block: false };
        }
      }
      
      // 3. 高风险工具 → 需要审批
      if (HIGH_RISK_TOOLS.has(toolName)) {
        // 更新任务状态为 waiting_approval
        config.onWaitingApproval(event);
        
        const approved = await config.onToolApprovalRequest({
          toolName,
          input: event.input,
        });
        
        return approved
          ? { block: false }
          : { block: true, reason: 'User denied' };
      }
      
      // 4. 未知工具 → 默认拒绝
      return { block: true, reason: `Unknown tool: ${toolName}` };
    });
  };
}
```

### P0-6: Token 自动刷新

**优先级**: P0  
**依赖**: P0-5  
**预计工时**: 3-4h

**目标**: 实例 token 15 分钟自动刷新

**实现方法**: (与原计划相同)

```typescript
// electron/instance-token-manager.ts
class InstanceTokenManager {
  private instanceToken: string | null = null;
  private expiresAt: number = 0;
  private refreshTimer: NodeJS.Timeout | null = null;
  private readonly REFRESH_BEFORE_MS = 5 * 60 * 1000;  // 提前 5 分钟
  
  async initialize(refreshToken: string, instanceId: string) {
    await this.refreshToken(refreshToken, instanceId);
    this.scheduleRefresh();
  }
  
  private async refreshToken(refreshToken: string, instanceId: string) {
    const { instanceToken, expiresIn } = await getInstanceToken({
      refreshToken,
      instanceId,
    });
    
    this.instanceToken = instanceToken;
    this.expiresAt = Date.now() + expiresIn * 1000;
  }
  
  private scheduleRefresh() {
    if (this.refreshTimer) clearTimeout(this.refreshTimer);
    
    const delay = Math.max(0, this.expiresAt - this.REFRESH_BEFORE_MS - Date.now());
    
    this.refreshTimer = setTimeout(async () => {
      await this.refreshToken(this.refreshToken, this.instanceId);
      this.scheduleRefresh();
    }, delay);
  }
  
  // 同步获取当前有效 token（before_provider_headers 使用）
  getValidTokenSync(): string {
    if (Date.now() >= this.expiresAt) {
      throw new Error('Token expired, refresh in progress');
    }
    return this.instanceToken!;
  }
}
```

**集成到 provider**:
```typescript
// pi-extension/provider.ts
const providerExtension: ExtensionFactory = (pi) => {
  pi.on('before_provider_headers', (event) => {
    const token = tokenManager.getValidTokenSync();
    event.headers['authorization'] = `Bearer ${token}`;
  });
};
```

**验收标准**:
- [ ] Token 提前 5 分钟自动刷新
- [ ] 刷新过程不影响正在执行的任务
- [ ] 长时间运行的任务（>15分钟）不会因 token 过期失败
